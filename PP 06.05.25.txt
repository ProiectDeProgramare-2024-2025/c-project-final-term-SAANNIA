#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#ifdef _WIN32
#include <direct.h>
#define CLEAR_COMMAND "cls"
#else
#include <unistd.h>
#define CLEAR_COMMAND "clear"

#define MAX_LENGTH 500

#define RED     "\033[1;31m"
#define GREEN   "\033[1;32m"
#define YELLOW  "\033[1;33m"
#define BLUE    "\033[1;34m"
#define RESET   "\033[0m"


int esteValidNumeSauPrenume(const char* s) {
    for (int i = 0; s[i]; i++) {
        if (!isalpha(s[i])) return 0;
    }
    return strlen(s) > 0;
}

int esteTextValid(const char* s) {
    for (int i = 0; s[i]; i++) {
        if (!isspace(s[i])) return 1;
    }
    return 0;
}


typedef struct {
    char nume[50];
    char prenume[50];
    char telefon[20];
    char experienta[200];
} CV;


void afiseazaCVs();

void clearScreen() {
    system(CLEAR_COMMAND);
}

void adaugaCV() {
    clearScreen();
    CV cv;

    printf("=== Adauga CV ===\n");

    do {
    printf("Nume (doar litere, fara cifre): ");
    scanf("%49s", cv.nume);
    } while (!esteValidNumeSauPrenume(cv.nume));

    do {
    printf("Prenume (doar litere, fara cifre): ");
    scanf("%49s", cv.prenume);
    } while (!esteValidNumeSauPrenume(cv.prenume));



    do {
        printf("Telefon (doar cifre): ");
        scanf("%19s", cv.telefon);

        int valid = 1;
        for (int i = 0; cv.telefon[i] != '\0'; i++) {
            if (!isdigit(cv.telefon[i])) {
                valid = 0;
                break;
            }
        }

        if (!valid) {
            printf("Numar invalid! Introdu doar cifre.\n");
        } else {
            break;
        }
    } while (1);

    getchar();
    do {
    printf("Experienta: ");
    fgets(cv.experienta, sizeof(cv.experienta), stdin);
    cv.experienta[strcspn(cv.experienta, "\n")] = 0;
    } while (!esteTextValid(cv.experienta));


    FILE *file = fopen("cvs.txt", "a");
    if (!file) {
        printf("Eroare la deschiderea fișierului!\n");
        return;
    }


    fprintf(file, "Nume: %s\n", cv.nume);
    fprintf(file, "Prenume: %s\n", cv.prenume);
    fprintf(file, "Telefon: %s\n", cv.telefon);
    fprintf(file, "Experienta: %s\n", cv.experienta);
    fprintf(file, "=====\n");

    fclose(file);

    printf("CV-ul a fost salvat cu succes!\nApasa Enter pentru a continua...");
    while (getchar() != '\n');
    clearScreen();
}

}
